  import json
  from functools import partial

  def load_scores(retrieval_type, memory_type):
      analysis_dir = f"../../analysis/bm25/{retrieval_type}/exp5/"
      file_name = f"{memory_type}.json"

      with open(analysis_dir + file_name, "r") as file:
          data = json.load(file)
      return data

  load_memory_boundary_fn = partial(load_scores, memory_type='Memory Boundary')
  bm25 = load_memory_boundary_fn('bm25')
  embeddings = load_memory_boundary_fn('embeddings')

  print(json.dumps(bm25['Omission'], indent=4), '\n', json.dumps(embeddings['Omission'], indent=4))
