  import pandas as pd
  import json
  import os

  scores_dir = "../../scores/bm25/"
  def load_score(exp_dir):
      dired = scores_dir + f"{exp_dir}/"
      if len(os.listdir(dired)) == 1:
          score_file = os.listdir(dired)[0]
          with open(dired + score_file, "r") as file:
              scores = json.load(file)
          return scores
      return

  def load_scores(scores_dir):
      exp_scores = {}
      per_persona_results = {}
      for exp_dir in os.listdir(scores_dir):
          score_path = scores_dir + exp_dir
          if os.path.isdir(score_path) and (not exp_dir.startswith('_')):
              scores = load_score(exp_dir)
              if scores is not None:
                  exp_scores[exp_dir] = {
                      k.split('_')[0]: v
                      for k, v in scores.pop('overall_score').items()
                      if "(all)" in k
                  }
      return exp_scores

  def make_table(scores: dict):
      score_table = {"Correct": {}, "Hallucination": {}, "Omission": {}}
      for exp in scores.keys():
          for key, value in scores[exp].items():
              score_table[key.capitalize()][exp] = value * 100
      return score_table

  scores = load_scores(scores_dir)
  score_table = make_table(scores)
  scores_df = pd.DataFrame(score_table)
  scores_df = scores_df.sort_index()
  print(tabulate(scores_df, headers='keys', tablefmt='orgtbl', floatfmt='.2f'))

#+NAME: label-dist-table
#+header: :prologue from tabulate import tabulate
#+begin_src python :results output table raw :exports results
  import matplotlib.pyplot as plt
  import pandas as pd
  import json
  import os

  scores_dir = "../../scores/bm25/"
  def load_score(exp_dir):
      dired = scores_dir + f"{exp_dir}/"
      if len(os.listdir(dired)) == 1:
          score_file = os.listdir(dired)[0]
          with open(dired + score_file, "r") as file:
              scores = json.load(file)
          return scores
      return

  def load_results(scores_dir):
      exp_scores = {}
      per_persona_results = {}
      for exp_dir in sorted(os.listdir(scores_dir)):
          score_path = scores_dir + exp_dir
          if os.path.isdir(score_path) and (not exp_dir.startswith('_')):
              scores = load_score(exp_dir)
              if scores is not None:
                  per_persona_results[exp_dir] = scores.pop('per_persona_results')
      return per_persona_results

  results = load_results(scores_dir)

  def per_type_results_df(results):
      per_type_results = {"Correct": {}, "Hallucination": {}, "Omission": {}}
      per_type_count = {}
      for result in results:
          question_type = result['question_type']
          result_type = result['result_type']
          if per_type_count.get(question_type) is None:
              per_type_count[question_type] = 0
          if per_type_results[result_type].get(question_type) is None:
              per_type_results[result_type][question_type] = 0
              
          per_type_results[result_type][question_type] += 1
          per_type_count[question_type] += 1

      per_type_normalized_results = {}
      for key, value in per_type_results.items():
          per_type_normalized_results[f"{key} (%)"] = {
              k: round(v / per_type_count[k] * 100, 2)
              if v else v
              for k, v in value.items()
          }

      per_type_normalized_results["Total"] = per_type_count
          
      df = pd.DataFrame(per_type_normalized_results)
      df = df.sort_values(by='Total', ascending=False)
      return df

  tables = ["*** 결과"]
  for exp, result in results.items():
      df = per_type_results_df(result)
      table = tabulate(df, headers='keys', tablefmt='orgtbl')

      tables.append(f"**** {exp}\n{table}")

  print("\n".join(tables).strip())
#+end_src
