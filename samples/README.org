결과 종류 3개 (Correct, Hallucination, Omission) 있어 각 종류가 12개 샘플링합니다.
#+begin_src python
  def per_type_scores(score_file):
    per_sample_scores = load_scores(score_file)['per_persona_results']
    result_types = {'Correct': [], 'Hallucination': [], 'Omission': []}
    for score in per_sample_scores:
        result_types[score['result_type']].append(score)

    return result_types

  def sample_results(score_file):
    random.seed(42)
    per_type_results_samples = {}
    results = per_type_results[score_file]
    for key in results:
        per_type_results_samples[key] = random.sample(results[key], 12)
    return per_type_results_samples
#+end_src

12개 설정하는 이유는 한명씩 나이브 메모리 시스템 결과를 ={"Correct": 3, "Hallucination": 3, "Omission": 3}= 총 9개 샘플을 살펴보도록 하기 때문입니다. 에이전틱 메모리 시스템과 싱크할 수 있도록 인덱싱도 합니다.

   def get_index(sample, score_file):
     per_sample_scores = load_scores(score_file)['per_persona_results']
     subset, _ = parse_settings(score_file)
     idx = per_sample_scores.index(sample)
     length = 0
     dataset = [json.loads(line) for line in open(f'dataset/HaluMem-{subset.capitalize()}.jsonl', 'r').readlines()]
     for i, persona in enumerate(dataset):
         q_sessions = [session['questions'] if session.get('questions') else False for session in persona['sessions']]
         for j, session in enumerate(q_sessions):
             if session:
                 length += len(session)
                 if length > idx:
                     ind = length - (length - len(session)) - (length - idx)
                     return i, j, ind
